# Proto App Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Proto — a Flutter Android app that silently monitors YouTube via Accessibility Service and blocks distracting content during timed study sessions.

**Architecture:** Flutter + Riverpod owns all UI, session timer, and user profile data. Kotlin owns the Accessibility Service, screen overlay, and Foreground Service. A thin MethodChannel bridge (`proto/session`) carries commands from Flutter to Kotlin at session start/stop only; an EventChannel (`proto/events`) streams blocked-count increments back to Flutter in real time.

**Tech Stack:** Flutter 3.x, Dart, Riverpod 2.x, go_router, shared_preferences, google_fonts; Kotlin, Android Accessibility Service, SYSTEM_ALERT_WINDOW overlay, Room DB, Android Foreground Service.

```kotlin
package com.proto.classifier

enum class ContentType { EDUCATIONAL, NON_EDUCATIONAL, SHORTS }

data class ClassifierConfig(
    val allowlist: List<String>,
    val keywords: List<String>,
)

object VideoClassifier {

    fun classify(cardText: String, config: ClassifierConfig): ContentType {
        val lower = cardText.lowercase()

        // Hard-block Shorts indicators
        if (isShorts(lower)) return ContentType.SHORTS

        // Allowlist check — channel name match
        val matchesAllowlist = config.allowlist.any { channel ->
            lower.contains(channel.lowercase())
        }
        if (matchesAllowlist) return ContentType.EDUCATIONAL

        // Keyword heuristic
        val matchesKeyword = config.keywords.any { keyword ->
            lower.contains(keyword.lowercase())
        }
        if (matchesKeyword) return ContentType.EDUCATIONAL

        return ContentType.NON_EDUCATIONAL
    }

    private fun isShorts(lower: String): Boolean {
        return lower.contains("shorts") ||
            lower.contains("#shorts") ||
            lower.contains("youtube shorts")
    }
}
```

```kotlin
package com.proto.services

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Intent
import android.os.IBinder
import androidx.core.app.NotificationCompat
import com.proto.MainActivity
import com.proto.proto.R

class ProtoForegroundService : Service() {

    companion object {
        const val CHANNEL_ID = "proto_study_mode"
        const val NOTIFICATION_ID = 1001
        const val ACTION_START = "proto.START"
        const val ACTION_STOP = "proto.STOP"
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_START -> startForeground(NOTIFICATION_ID, buildNotification())
            ACTION_STOP -> {
                stopForeground(STOP_FOREGROUND_REMOVE)
                stopSelf()
            }
        }
        return START_NOT_STICKY
    }

    override fun onBind(intent: Intent?): IBinder? = null

    private fun buildNotification(): Notification {
        createChannel()
        val openApp = PendingIntent.getActivity(
            this, 0,
            Intent(this, MainActivity::class.java),
            PendingIntent.FLAG_IMMUTABLE,
        )
        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle(getString(R.string.notification_title))
            .setContentText(getString(R.string.notification_text))
            .setContentIntent(openApp)
            .setOngoing(true)
            .setPriority(NotificationCompat.PRIORITY_LOW)
            .build()
    }

    private fun createChannel() {
        val channel = NotificationChannel(
            CHANNEL_ID,
            getString(R.string.notification_channel_name),
            NotificationManager.IMPORTANCE_LOW,
        )
        getSystemService(NotificationManager::class.java).createNotificationChannel(channel)
    }
}
```

```kotlin
class OverlayManager(private val context: Context) {

    private val mainHandler = Handler(Looper.getMainLooper())
    private val activeOverlays = mutableListOf<View>()

    /** Show a translucent dark overlay over a specific screen region.

        mainHandler.post {
```

```kotlin
package com.proto.services

import android.accessibilityservice.AccessibilityService
import android.content.Intent
import android.view.accessibility.AccessibilityEvent
import android.view.accessibility.AccessibilityNodeInfo
import com.proto.bridge.EventBridge
import com.proto.classifier.ClassifierConfig
import com.proto.classifier.ContentType
import com.proto.classifier.VideoClassifier
import com.proto.overlay.OverlayManager

class ProtoAccessibilityService : AccessibilityService() {

    companion object {
        var isRunning = false
        var config: ClassifierConfig? = null
    }

    private val overlayManager by lazy { OverlayManager(applicationContext) }
    private var blockedCount = 0

    override fun onServiceConnected() {
        isRunning = true
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        val cfg = config ?: return
        if (!isRunning) return

        val packageName = event?.packageName?.toString() ?: return
        if (packageName != "com.google.android.youtube") return

        when (event.eventType) {
            AccessibilityEvent.TYPE_WINDOW_CONTENT_CHANGED,
            AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED -> {
                processYouTubeScreen(cfg)
            }
        }
    }

    private fun processYouTubeScreen(cfg: ClassifierConfig) {
        val root = rootInActiveWindow ?: return
        overlayManager.clearOverlays()

        // Collect all visible text nodes that resemble video cards
        val videoNodes = mutableListOf<AccessibilityNodeInfo>()
        collectVideoNodes(root, videoNodes)

        videoNodes.forEach { node ->
            val text = extractNodeText(node)
            when (VideoClassifier.classify(text, cfg)) {
                ContentType.NON_EDUCATIONAL -> {
                    overlayCardNode(node)
                    blockedCount++
                    EventBridge.sendBlockedCount(blockedCount)
                }
                ContentType.SHORTS -> {
                    // Shorts are handled by Flutter overlay via EventBridge signal
                    EventBridge.sendBlockedCount(blockedCount)
                }
                ContentType.EDUCATIONAL -> { /* allow */ }
            }
        }
    }

    private fun collectVideoNodes(
        node: AccessibilityNodeInfo,
        result: MutableList<AccessibilityNodeInfo>,
        depth: Int = 0,
    ) {
        if (depth > 8) return
        // YouTube video cards typically have viewId containing "video_card" or "reel"
        val viewId = node.viewIdResourceName ?: ""
        if (viewId.contains("video_card") || viewId.contains("reel_item") ||
            viewId.contains("compact_video")) {
            result.add(node)
            return
        }
        for (i in 0 until node.childCount) {
            node.getChild(i)?.let { collectVideoNodes(it, result, depth + 1) }
        }
    }

    private fun extractNodeText(node: AccessibilityNodeInfo): String {
        val sb = StringBuilder()
        fun traverse(n: AccessibilityNodeInfo) {
            n.text?.let { sb.append(it).append(' ') }
            n.contentDescription?.let { sb.append(it).append(' ') }
            for (i in 0 until n.childCount) { n.getChild(i)?.let { traverse(it) } }
        }
        traverse(node)
        return sb.toString()
    }

    private fun overlayCardNode(node: AccessibilityNodeInfo) {
        val rect = android.graphics.Rect()
        node.getBoundsInScreen(rect)
        overlayManager.showCardOverlay(rect.left, rect.top, rect.width(), rect.height())
    }

    override fun onInterrupt() {
        isRunning = false
        overlayManager.cleanup()
    }

    override fun onDestroy() {
        isRunning = false
        config = null
        overlayManager.cleanup()
        super.onDestroy()
    }
}
```

```kotlin
class SessionBridge(private val context: Context) : MethodChannel.MethodCallHandler {

    private val db by lazy {
        Room.databaseBuilder(context, ProtoDatabase::class.java, "proto_db").build()
    }
    private val scope = CoroutineScope(Dispatchers.IO)

    override fun onMethodCall(call: MethodCall, result: MethodChannel.Result) {
        when (call.method) {
            "startSession" -> {
                val durationMinutes = call.argument<Int>("durationMinutes") ?: 25
                val allowlist = call.argument<List<String>>("allowlist") ?: emptyList()
                val keywords = call.argument<List<String>>("keywords") ?: emptyList()
                ProtoAccessibilityService.config = ClassifierConfig(allowlist, keywords)
                startForegroundService()
                result.success(null)
            }
            "stopSession" -> {
                stopForegroundService()
                ProtoAccessibilityService.config = null
                result.success(null)
            }
            "saveSession" -> {
                val record = SessionRecord(
                    startTime = call.argument<Long>("startTime") ?: 0L,
                    endTime = call.argument<Long>("endTime") ?: 0L,
                    durationMinutes = call.argument<Int>("durationMinutes") ?: 0,
                    blockedCount = call.argument<Int>("blockedCount") ?: 0,
                )
                scope.launch { db.sessionDao().insert(record) }
                result.success(null)
            }
            "getSessionHistory" -> {
                scope.launch {
                    val records = db.sessionDao().getAll()
                    val mapped = records.map {
                        mapOf(
                            "id" to it.id,
                            "startTime" to it.startTime,
                            "endTime" to it.endTime,
                            "durationMinutes" to it.durationMinutes,
                            "blockedCount" to it.blockedCount,
                        )
                    }
                    CoroutineScope(Dispatchers.Main).launch { result.success(mapped) }
                }
            }
            "checkPermissions" -> {
                result.success(mapOf(
                    "accessibility" to ProtoAccessibilityService.isRunning,
                    "overlay" to Settings.canDrawOverlays(context),
                    "notification" to true, // simplified; real check needs NotificationManager
                ))
            }
            "openAccessibilitySettings" -> {
                context.startActivity(Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS).apply {
                    flags = Intent.FLAG_ACTIVITY_NEW_TASK
                })
                result.success(null)
            }
            "openOverlaySettings" -> {
                context.startActivity(Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION).apply {
                    flags = Intent.FLAG_ACTIVITY_NEW_TASK
                })
                result.success(null)
            }
            else -> result.notImplemented()
        }
    }

    private fun startForegroundService() {
        context.startForegroundService(
            Intent(context, ProtoForegroundService::class.java).apply {
                action = ProtoForegroundService.ACTION_START
            }
        )
    }

    private fun stopForegroundService() {
        context.startService(
            Intent(context, ProtoForegroundService::class.java).apply {
                action = ProtoForegroundService.ACTION_STOP
            }
        )
    }
```

```dart
class SessionNotifier extends StateNotifier<SessionState> {
  static const _method = MethodChannel(AppConstants.methodChannel);
  static const _events = EventChannel(AppConstants.eventChannel);

  Timer? _timer;
  StreamSubscription? _eventSub;

  SessionNotifier(int defaultMins) : super(SessionState(durationMinutes: defaultMins));

  void setDuration(int minutes) {
    if (state.status != SessionStatus.idle) return;
    state = state.copyWith(durationMinutes: minutes);
  }

  Future<void> start(List<String> allowlist) async {
    state = state.copyWith(
      status: SessionStatus.active,
      elapsedSeconds: 0,
      blockedCount: 0,
      startTime: DateTime.now(),
    );

    await _method.invokeMethod(AppConstants.methodStartSession, {
      'durationMinutes': state.durationMinutes,
      'allowlist': allowlist,
      'keywords': EduKeywords.terms,
    });

    _eventSub = _events.receiveBroadcastStream().listen((event) {
      if (event is Map && event['type'] == 'blockedCount') {
        state = state.copyWith(blockedCount: event['count'] as int);
      }
    });

    _timer = Timer.periodic(const Duration(seconds: 1), (_) {
      if (state.elapsedSeconds >= state.totalSeconds) {
        stop();
      } else {
        state = state.copyWith(elapsedSeconds: state.elapsedSeconds + 1);
      }
    });
  }

  Future<SessionState> stop() async {
    _timer?.cancel();
    _eventSub?.cancel();
    final endTime = DateTime.now();
    final finalState = state;

    await _method.invokeMethod(AppConstants.methodStopSession);
    await _method.invokeMethod(AppConstants.methodSaveSession, {
      'startTime': finalState.startTime?.millisecondsSinceEpoch ?? 0,
      'endTime': endTime.millisecondsSinceEpoch,
      'durationMinutes': finalState.durationMinutes,
      'blockedCount': finalState.blockedCount,
    });

    state = SessionState(durationMinutes: finalState.durationMinutes);
    return finalState;
  }

  @override
  void dispose() {
    _timer?.cancel();
    _eventSub?.cancel();
    super.dispose();
```

## Known Constraints

- **Accessibility Service node IDs change with YouTube app updates.** If overlays stop working after a YouTube update, re-inspect node IDs with Android Studio's Layout Inspector or `adb shell uiautomator dump`.
- **SYSTEM_ALERT_WINDOW** requires user grant on Android 6+. The permission sheet handles this.

- **Foreground Service `specialUse` type** requires the `PROPERTY_SPECIAL_USE_FGS_SUBTYPE` property in Manifest (Android 14+). Already included.
